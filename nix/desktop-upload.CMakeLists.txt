cmake_minimum_required(VERSION 3.20)
project(OpenCloudUpload LANGUAGES CXX)

include(GNUInstallDirs)
include(GenerateExportHeader)
include(VERSION.cmake)

find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Network)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)

configure_file(src/libsync/common/version.cpp.in generated/version.cpp @ONLY)

add_library(opencloud-upload STATIC
    src/libsync/abstractnetworkjob.cpp
    src/libsync/accessmanager.cpp
    src/libsync/account.cpp
    src/libsync/appprovider.cpp
    src/libsync/capabilities.cpp
    src/libsync/cookiejar.cpp
    src/libsync/httplogger.cpp
    src/libsync/jobqueue.cpp
    src/libsync/discovery.cpp
    src/libsync/localdiscoverytracker.cpp
    src/libsync/owncloudpropagator.cpp
    src/libsync/propagateupload.cpp
    src/libsync/propagateuploadtus.cpp
    src/libsync/propagateuploadv1.cpp
    src/gui/folder.cpp
    src/gui/folderwatcher.cpp
    src/gui/folderwatcher_linux.cpp
    src/libsync/creds/abstractcredentials.cpp
    src/libsync/common/checksumalgorithms.cpp
    src/libsync/common/chronoelapsedtimer.cpp
    src/libsync/common/utility.cpp
    generated/version.cpp
)

set_target_properties(opencloud-upload PROPERTIES
    POSITION_INDEPENDENT_CODE ON
    CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN ON
)
target_compile_definitions(opencloud-upload PUBLIC
    OPENCLOUD_BUILT_AS_STATIC
    QT_NO_KEYWORDS
    WITH_HTTP2=1
)
target_compile_definitions(opencloud-upload PRIVATE
    QT_NO_CAST_TO_ASCII
    QT_NO_CAST_FROM_ASCII
    QT_NO_URL_CAST_FROM_STRING
    QT_NO_CAST_FROM_BYTEARRAY
    QT_USE_QSTRINGBUILDER
    QT_MESSAGELOGCONTEXT
    QT_NO_FOREACH
    QT_DISABLE_DEPRECATED_BEFORE=0x060200
    QT_NO_CONTEXTLESS_CONNECT
)
target_compile_options(opencloud-upload PRIVATE -ffunction-sections -fdata-sections)
target_include_directories(opencloud-upload PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/src
    ${CMAKE_CURRENT_SOURCE_DIR}/src/libsync
    ${CMAKE_CURRENT_SOURCE_DIR}/src/gui
    ${CMAKE_CURRENT_BINARY_DIR}/generated
    ${CMAKE_CURRENT_BINARY_DIR}/generated/libsync
)
target_link_libraries(opencloud-upload PUBLIC Qt6::Core Qt6::Gui Qt6::Network)

generate_export_header(opencloud-upload
    EXPORT_MACRO_NAME OPENCLOUD_SYNC_EXPORT
    EXPORT_FILE_NAME generated/libsync/opencloudsynclib.h
    STATIC_DEFINE OPENCLOUD_BUILT_AS_STATIC
)

# src/gui headers include "gui/opencloudguilib.h"; the upstream build generates it
# from the opencloudgui target, which we do not build. Alias it onto our own macro.
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/generated/gui/opencloudguilib.h
    "#pragma once\n#include \"libsync/opencloudsynclib.h\"\n#define OPENCLOUD_GUI_EXPORT OPENCLOUD_SYNC_EXPORT\n")

configure_file(opencloud-upload.pc.in opencloud-upload.pc @ONLY)
install(TARGETS opencloud-upload ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY src/libsync DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/opencloud-upload
    FILES_MATCHING PATTERN "*.h")
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/generated/libsync/opencloudsynclib.h
    DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/opencloud-upload/libsync)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/opencloud-upload.pc
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
install(FILES COPYING DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/opencloud-upload)
